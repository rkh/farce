Includes:
Extension.new
Defined in:

Overview

Backports for Ractor and related functionality to older Ruby versions.

Activating Backports

There are three ways of using these backports:

  • Activating them globally via: Backports.install
  • Using them within a specific scope via refinements: using Farce::Backports
  • Calling methods on Farce::Backports directly

Only the first method will also define the global constants Ractor and Ractor::Port if they are not natively available.

Examples:

Activating the backports globally

Farce::Backports.install
port = Ractor.current.default_port

Using the backports within a specific namespace via refinements

# Within a namespace
module MyNamespace
  using Farce::Backports

  def self.send_to(ractor_or_port, message)
    port = ractor_or_port.default_port if ractor_or_port.is_a?(Ractor)
    port.send(message)
  end
end

Activating the backports for a specific file

# top level of the file
using Farce::Backports

ractor  = Ractor.new { sleep 10 }
monitor = Ractor.new { puts "exit status: #{receive}" }

monitor.monitor(ractor.default_port)
monitor.join

Activating the backports for a given proc (Ruby 4.1+ only)

callback = ->(ractor) { ractor.default_port }
callback.refined(Farce::Backports).call(Ractor.current)

Calling methods on Farce::Backports directly

port = Farce::Backports.default_port(Ractor.current)

module Backports
  # @!visibility private
  Extension = Farce::Extension.new
  private_constant :Extension
  include Extension

  if Internal.native_ractors?
    if needs_timeouts = ::Ractor.instance_method(:receive).arity.zero?
      Extension.define(::Ractor) do
        def receive(timeout: nil)
          return super() if timeout.nil?
          Internal::RactorSelector.current.ractor_receive(self, timeout:)
        end
      end

      Extension.define(::Ractor.singleton_class) do
        def select(*sources, timeout: nil, **)
          # The selector helper must reach Ruby's native wait without reentering itself.
          return super(*sources, **) if timeout.nil? && (Internal.native_ports? || sources.all?(::Ractor))
          Farce::Ractor.select(*sources, timeout:, **)
        end
      end

      def self.receive(ractor_or_port, timeout: nil)
        ractor_or_port = default_port(ractor_or_port) if ractor_or_port.is_a?(Ractor)
        return ractor_or_port.receive if timeout.nil?
        return ractor_or_port.receive(timeout:) unless Internal.native_ports?
        Internal::RactorSelector.current.ractor_receive(ractor_or_port, timeout:)
      end
    end

    if Internal.native_ports?
      unless ::Ractor.method_defined?(:alive?)
        Extension.define(::Ractor) { def alive? = Backports.alive?(self) }

        def self.alive?(ractor)
          ractor.monitor(NULL_PORT)
        ensure
          ractor.unmonitor(NULL_PORT)
        end
      end

      if needs_timeouts
        Extension.define(::Ractor::Port) do
          def receive(timeout: nil)
            return super() if timeout.nil?
            Internal::RactorSelector.current.ractor_receive(self, timeout:)
          end
        end
      end
    else
      Extension.define(::Ractor) do
        def alive?          = Backports.alive?(self)
        def default_port    = Backports.default_port(self)
        def join            = Backports.join(self)
        def monitor(port)   = Backports.monitor(self, port)
        def unmonitor(port) = Backports.unmonitor(self, port)
        def value           = Backports.value(self)
      end

      def self.alive?(ractor)
        return true if Ractor.main == ractor
        state = Internal::RactorState.for(ractor)
        return state.alive? if state
        !ractor.inspect.end_with?("terminated>")
      end

      def self.default_port(ractor) = Farce::Ractor::Port.new(ractor)

      def self.join(ractor)
        Internal::RactorState.for(ractor)&.join
        ractor
      end

      def self.monitor(ractor, port)
        return true if Ractor.main == ractor

        state = Internal::RactorState.for(ractor)
        return state.monitor(port) if state

        raise Ractor::Error, "#{ractor.inspect} cannot be monitored" unless ractor.inspect.end_with?("terminated>")

        success = true

        begin
          ractor.take
        rescue Ractor::ClosedError
          # no-op
        rescue Ractor::RemoteError
          success = false
        end

        port.send(success ? :exited : :aborted)
        false
      end

      def self.unmonitor(ractor, port)
        Internal::RactorState.for(ractor)&.unmonitor(port)
        ractor
      end

      def self.value(ractor)
        Internal::RactorState.for(ractor)&.join
        values = ::Ractor.store_if_absent("Ractor#value") { ObjectSpace::WeakKeyMap.new }
        entry  = values[ractor] ||= [Mutex.new]
        entry.first.synchronize { entry << ractor.take if entry.size == 1 }
        entry[1]
      end
    end
  end

  %i[alive? default_port join monitor unmonitor value].each do |method|
    next if respond_to?(method)
    class_eval <<~RUBY, __FILE__, __LINE__ + 1
      def self.#{method}(ractor, ...) = ractor.#{method}(...)
    RUBY
  end

  unless respond_to?(:receive)
    def self.receive(ractor_or_port, timeout: nil)
      ractor_or_port = default_port(ractor_or_port) if ractor_or_port.is_a?(Ractor)
      timeout.nil? ? ractor_or_port.receive : ractor_or_port.receive(timeout:)
    end
  end

  def self.select(...) = Farce::Ractor.select(...)

  # Installs the Farce backports for Ractor and Ractor::Port globally.
  def self.install
    Extension.install do
      if Internal.native_ractors?
        ::Ractor.const_set(:Port, Farce::Ractor::Port) unless Internal.native_ports?
      else
        Object.const_set(:Ractor, Farce::Ractor)
      end
    end
  end
end

        entry.first.synchronize { entry << ractor.take if entry.size == 1 }
        entry[1]
      end
    end
  end

    next if respond_to?(method)
    class_eval <<~RUBY, __FILE__, __LINE__ + 1
      def self.#{method}(ractor, ...) = ractor.#{method}(...)
    RUBY
  end

  unless respond_to?(:receive)
    def self.receive(ractor_or_port, timeout: nil)
      ractor_or_port = default_port(ractor_or_port) if ractor_or_port.is_a?(Ractor)
      timeout.nil? ? ractor_or_port.receive : ractor_or_port.receive(timeout:)
    end
  end

  def self.select(...) = Farce::Ractor.select(...)

  def self.install

      else

      end
    end
  end
