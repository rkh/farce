Overview

A publisher subscription that can be paused or permanently canceled. Disabling skips new messages. Unsubscribing drains an existing callback queue before stopping it.

def self.define(*classes, &)
  raise LocalJumpError, "no block given" unless block_given?
  definition = Internal.prepare_method_definition(&)
  classes.each do |klass|
    REGISTER.define(klass) do
      define_method(:deliver, definition)
      private :deliver
    end
  end
end

def unsubscribe

  @map.delete_pair(key, self)
  subscriber.seal if @callback
