Overview

A unit of work that executes once and retains its result or failure.

# You can create and schedule a task with one call:
task = Farce::Task.schedule { 1 + 2 }
task.value # => 3

# There is also a shorthand for running a task on the global Ractor pool:
task = Farce::Task.in_parallel { sleep 1 }
task.join
task.alive? # => false

# schedule accepts a custom scheduler:
scheduler = Farce::Scheduler.create
task      = Farce::Task.schedule(scheduler:) { 1 + 2 }

# or alternatively you can manually create a task and hand it to the scheduler
task = Farce::Task.new { 1 + 2 }
scheduler.schedule(&task)

# or execute it manually
task = Farce::Task.new { Ractor.current.name.upcase }
Ractor.new(task, name: "Herbert") { it.call }
task.value # => "HERBERT"

Constructor Details

#initialize(*args, mode: :smart) {|*args| ... } ⇒ Task

Creates a new task with the given arguments and callback.

This does not schedule the task for execution, nor does it associate it with any executor.

In fact, the task object itself is a proc-like object, that can be called like a regular proc, but will track its execution state and result.

Valid modes are:

  • :smart - Applies smart rules or a dedicated proxy. Raises for unsupported values. This is the default for most mode-backed classes.
  • :smart_raise - Applies smart rules or a dedicated proxy, with a raising fallback.
  • :smart_copy - Applies smart rules or a dedicated proxy, then falls back to copying or Marshal. This is the default for Port.
  • :smart_move - Applies smart rules or a dedicated proxy, then falls back to moving.
  • :copy - The value will be copied between Ractors.
  • :marshal - The value will be serialized immediately and reconstructed once per Ractor with Ruby's Marshal protocol. Contents must support Marshal. Nested shareable values may be reconstructed too.
  • :make_shareable - The value will be made Ractor-shareable using Ractor.make_shareable.
  • :move - The value will be moved between Ractors. This saves memory compared to copying, and supports values that can't be copied but moved (like IO objects). However, the value will no longer be accessible on the Ractor that pushed it.
  • :mutable - A Mutable instance will be created for the value. This isn't done recursively and thus will fail for nested unshareable values.
  • :local - The value will be kept local to the Ractor that pushed it. Another ractor trying to receive it will get an error. Useful for usage contained within a single Ractor.
  • :proxy - The value will be wrapped in a Farce::Proxy that executes calls in the original Ractor.
  • :raise - An error will be raised if the value is not Ractor-shareable. Useful for enforcing shareability.
  • :dedup - The value will be deduplicated using Farce.dedup, then made Ractor-shareable. This may update and freeze the original. Already-shareable values pass through unchanged.
  • :shareable_copy - The value will be copied and the copy will be made Ractor-shareable.

Examples:

Using a Task without an Executor

task = Farce::Task.new(1, 2) { |a, b| a + b }

task.alive?       # => true
task.call == task # => true
task.alive?       # => false
task.value        # => 3

Passing a task to an executor

task = Farce::Task.new { puts "Hello, world!" }
Farce.in_parallel(&task)
task.join # waits for the task to complete

def initialize(*args, mode: :smart, &callback)
  raise ArgumentError, "invalid mode: #{mode}" unless MODES.include?(mode)
  raise LocalJumpError, "no callback provided" unless block_given?

  @mode   = mode
  @ractor = Strict::Atom.new
  @state  = Strict::Atom.new(:initialized)

  if mode == :local
    @arguments    = args
    @callback     = callback
    @manager      = nil
    @ractor.value = Ractor.current
    @result       = Unshared::Atom.new
  else
    @arguments = Strict::Vector.new
    @callback  = Ractor.shareable?(callback) ? callback : Ractor.shareable_proc(&callback)
    @manager   = ModeManager.new(mode: mode)
    @result    = Strict::Atom.new
    args.each { add_argument(it) }
  end

  __freeze__
end

Adds an argument to the task. Can only be called before the task is frozen and before #call is invoked.

Allows for arguments to use different modes than the return value (or other arguments). Mode is ignored for local tasks.

def add_argument(argument, mode: nil)
  check_frozen!

  unless local?
    raise ArgumentError, "invalid mode: #{mode}" unless mode.nil? || MODES.include?(mode)
    raise InvalidStateError, "cannot add local argument to a non-local task" if mode == :local && !claim_for_ractor?
    argument = @manager.wrap(argument, mode:)
  end

  @arguments << argument
  self
end

def call
  raise InvalidStateError, "task already claimed by another ractor" unless claim_for_ractor?
  raise InvalidStateError, "call may only be called once" unless @state.compare_and_set(:initialized, :running)
  begin
    arguments     = local? ? @arguments.freeze : @arguments.freeze.to_a.map! { @manager.unwrap(it) }
    result        = @callback.call(*arguments)
    @result.value = local? ? result : @manager.wrap(result)
    @state.value  = :finished
  rescue Exception => e # rubocop:disable Lint/RescueException
    # Every callback failure must release waiters, including ScriptError and SystemExit.
    begin
      @result.value = e
    rescue Ractor::Error
      @result.value = [RuntimeError, -"#{e.class}: #{e.message}", Ractor.make_shareable(e.backtrace)].freeze
    end
    @state.value = :error
  ensure
    if @state.value == :running
      @result.value = [RuntimeError, -"task execution aborted"].freeze
      @state.value  = :error
    end
  end

  self
end

def value(timeout = nil)
  case @state.wait_while(timeout:) { alive_state?(it) }
  when :finished then local? ? @result.value : @manager.unwrap(@result.value)
  when :error    then raise(*@result.value)
  end
end
